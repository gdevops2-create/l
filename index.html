#include <fcntl.h>
#include <unistd.h>

{
    int source, destination;
    char buffer[100];

    source = open("source.txt", O_RDONLY);
    destination = open("destination.txt",
                       O_WRONLY | O_CREAT, 0644);

    close(source);

#include <unistd.h>
#include <stdlib.h>

int main()
{
    pid_t pid;

    pid = fork();

    if (pid < 0)
    {
        printf("Fork failed\n");
    }
    else if (pid == 0)
    {
        printf("Child process\n");
        printf("Child PID = %d\n", getpid());
        exit(0);
    }
    else
    {
        printf("Parent process\n");
        printf("Parent PID = %d\n", getpid());

        sleep(10);

        printf("Parent completed\n");
    }

    return 0;
}

#include <stdio.h>
#include <unistd.h>
#include <string.h>

int main()
{
    int fd[2];
    pid_t pid;

    char message[] = "Hello from Parent";
    char buffer[100];

    pipe(fd);

    pid = fork();

    if(pid == 0)
    {
        close(fd[1]);

        read(fd[0], buffer, sizeof(buffer));

        printf("Child received: %s\n", buffer);

        close(fd[0]);
    }
    else
    {
        close(fd[0]);

        write(fd[1], message, strlen(message) + 1);

        printf("Parent sent: %s\n", message);

        close(fd[1]);
    }

    return 0;
}
Result: Parent and child processes successfully communicated using an unnamed pipe.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <sys/wait.h>

int main()
{
    int shmid;
    char *shared_memory;

    shmid = shmget(1234, 1024, IPC_CREAT | 0666);

    if(shmid < 0)
    {
        printf("Shared memory creation failed\n");
        exit(1);
    }

    shared_memory = (char *)shmat(shmid, NULL, 0);

    if(shared_memory == (char *)-1)
    {
        printf("Shared memory attachment failed\n");
        exit(1);
    }

    if(fork() == 0)
    {
        sleep(1);

        printf("Child received: %s\n", shared_memory);

        shmdt(shared_memory);
    }
    else
    {
        strcpy(shared_memory, "Hello from Parent");

        printf("Parent sent: %s\n", shared_memory);

        wait(NULL);

        shmdt(shared_memory);

        shmctl(shmid, IPC_RMID, NULL);
    }

    return 0;
}

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main()
{
    int server_socket, client_socket;

    struct sockaddr_in server, client;

    socklen_t client_len = sizeof(client);

    char buffer[100];

    server_socket = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(8080);

    bind(server_socket,
         (struct sockaddr *)&server,
         sizeof(server));

    listen(server_socket, 5);

    printf("Server waiting for client...\n");

    client_socket = accept(
        server_socket,
        (struct sockaddr *)&client,
        &client_len
    );

    read(client_socket, buffer, sizeof(buffer));

    printf("Client says: %s\n", buffer);

    strcpy(buffer, "Hello from Server");

    write(client_socket,
          buffer,
          strlen(buffer) + 1);

    printf("Server sent: %s\n", buffer);

    close(client_socket);
    close(server_socket);

    return 0;
}
tcpclient.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main()
{
    int client_socket;

    struct sockaddr_in server;

    char buffer[100];

    client_socket = socket(
        AF_INET,
        SOCK_STREAM,
        0
    );

    server.sin_family = AF_INET;
    server.sin_port = htons(8080);
    server.sin_addr.s_addr =
        inet_addr("127.0.0.1");

    connect(
        client_socket,
        (struct sockaddr *)&server,
        sizeof(server)
    );

    strcpy(buffer, "Hello from Client");

    write(
        client_socket,
        buffer,
        strlen(buffer) + 1
    );

    printf("Client sent: %s\n", buffer);

    read(
        client_socket,
        buffer,
        sizeof(buffer)
    );

    printf("Server says: %s\n", buffer);

    close(client_socket);

    return 0;
}
